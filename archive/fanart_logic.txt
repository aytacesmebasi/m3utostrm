def _download_image(url, destination_path, callback=None):
    """
    Downloads an image from a URL to a destination path.

    Args:
        url (str): The URL of the image to download.
        destination_path (str): The local path to save the image.
        callback (function, optional): A function to report status.
    """
    if not url:
        return
    if os.path.exists(destination_path):
        if callback:
            callback(_("Already exists: {filename}").format(filename=os.path.basename(destination_path)))
        return
    try:
        if callback:
            callback(_("Downloading: {filename}").format(filename=os.path.basename(destination_path)))
        resp = requests.get(url, timeout=10, headers={'User-Agent': 'Mozilla/5.0'})
        if resp.status_code == 200:
            with open(destination_path, "wb") as f:
                f.write(resp.content)
    except Exception:
        pass
